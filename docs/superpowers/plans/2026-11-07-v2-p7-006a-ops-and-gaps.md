### 本计划作出并记录的决定（不需要用户再答，但列在这里以免被当成默认）

- **D-1 对账的落点 = 受绑定代码**，不是独立的不受绑定工具。依据见 T2 的「为什么不是独立工具」：拒绝必须**先于** `_commit`（`paper_evening.py:445-483`），而 `_commit` 只有晚间/结账路径能挡住；分歧必须随第 8 步的 journal `result` 立档（只有每日本写）；"到期订单集合"只有 `settlement.attempts` 说得准确（`attempt_outcomes` 已经把它们接到裁决上）。不受绑定的工具只能在事后报告，挡不住账簿推进。今天没有任何登记，受绑定代码免费。
- **D-2 `--show-book` 是 `scripts/daily_selection.py` 的只读模式**（与 `--factor-gaps` 同形），不是新的 `src/` 合约：它读的是既有快照的形状，渲染是脚本的职责；`src/` 一行不改。

- **D-5 G1 按"同一观测瞬间不重写"去重**，不改请求数（每个 `--year` 仍各抓一次整份登记簿）。请求本身是浪费（见 §7 第 2 条），但 G1 明确要求"请求数不变"，改动它需要另一个 issue。

def show_book(runtime_dir: Path) -> dict[str, Any]:
    """The paper book's head snapshot as a listing -- cash, every holding's lots and cost basis,
    the last mark -- read from the book's own state, writing nothing and asking the network
    nothing (`V2-P7-006a`).

    The snapshot directory is the book (`runtime.composition.build_storage`), and the reader is
    `FilePaperStateStore`, whose reads are total and create nothing (`storage/paper_state.py`):
    `build_storage` is deliberately not used, because it runs migrations and opens SQLite files,
    and a P6 day must not load the paper book at all. A store holding no snapshot is refused by
    name -- a missing book is never an empty one (`backtest.paper_state.held_head`).

    Marks are in the ledger's units (the real close times the adjustment ratio since the buy), so
    the document says so and is not read as real prices.
    """
    from openalpha_cn.backtest.paper_state import held_head
    from openalpha_cn.storage.paper_state import FilePaperStateStore
    from openalpha_cn.strategy_survivor_summary import SURVIVOR_WORDING

    store = FilePaperStateStore(runtime_dir.joinpath(*PAPER_STATE_DIRECTORY))
    snapshot = held_head(store)
    if snapshot is None:
        raise ValueError(
            "no paper book snapshot is held at all under "
            f"{runtime_dir.joinpath(*PAPER_STATE_DIRECTORY)}: a paper book is opened by saving an "
            "opening snapshot, never started from zero by a reader; nothing is created here"
        )
    book = snapshot.book
    positions = []
    for position in book.positions:
        mark = book.mark(position.subject)
        value = Decimal(position.quantity) * mark
        positions.append(
            {
                "ts_code": position.subject,
                "quantity": position.quantity,
                "cost_basis": format(position.cost_basis, "f"),
                "lots": [
                    {
                        "open_date": lot.open_date.isoformat(),
                        "quantity": lot.quantity,
                        "cost_basis": format(lot.cost_basis, "f"),
                    }
                    for lot in position.lots
                ],
                "mark": format(mark, "f"),
                "market_value": format(value, "f"),
                "unrealised": format(value - position.cost_basis, "f"),
            }
        )
    return {
        "kind": BOOK_VIEW_KIND,
        "session": snapshot.session.isoformat(),
        "session_index": snapshot.session_index,
        "snapshot_id": snapshot.snapshot_id,
        "config_digest": snapshot.config_digest,
        "first_decision_session": snapshot.first_decision_session,
        "last_settled_session": snapshot.last_settled_session,
        "cash": format(book.cash, "f"),
        "market_value": format(book.market_value, "f"),
        "equity": format(book.equity, "f"),
        "positions": positions,
        "pending_orders": len(snapshot.pending_orders),
        "marks": "ledger units (the real close times the adjustment ratio since the buy), not real closes",
        "wording": SURVIVOR_WORDING,
    }


def book_view_lines(view: Mapping[str, Any]) -> list[str]:
    """`show_book`'s document as the text an operator reads: one row per figure, one line per
    holding, one line per lot, and a last line saying nothing was written and nothing was asked.
    """
    lines = [
        f"kind               {view['kind']}",
        f"session            {view['session']} (session index {view['session_index']})",
        f"snapshot           {view['snapshot_id']}",
        f"config digest      {view['config_digest']}",
        f"first decided      {view['first_decision_session']}",
        f"last settled       {view['last_settled_session']}",
        f"cash               {view['cash']}",
        f"market value       {view['market_value']}",
        f"equity             {view['equity']}",
        f"positions          {len(view['positions'])}",
    ]
    for position in view["positions"]:
        lines.append(
            f"  {position['ts_code']:<12} {position['quantity']} shares, cost "
            f"{position['cost_basis']}, mark {position['mark']}, value "
            f"{position['market_value']}, unrealised {position['unrealised']}"
        )
        for lot in position["lots"]:
            lines.append(
                f"    lot            {lot['open_date']} {lot['quantity']} shares, cost "
                f"{lot['cost_basis']}"
            )
    lines += [
        f"pending orders     {view['pending_orders']}",
        f"marks              {view['marks']}",
        f"wording            {view['wording']}",
        "read-only          this listing wrote nothing and made no request",
    ]
    return lines

class Reconciliation: session, orders, divergences; reconciled; def document(self)

# paper_evening.py -- added keyword, defaulted (tests call these directly)

def run_catch_up(..., account: AccountRecord | None = None) -> CatchUpResult
# EveningResult/CatchUpResult gain `reconciliation: Reconciliation | None` (REQUIRED: four

# strategy_survivor_book.py -- same keyword on catch_up / settle_only / catch_up_again / evening

      runtime_dir: Path, *, session: date, explicit: Path | None
  ) -> AccountRecord | None:

      does not. Without it the day looks for `<runtime>/paper/account-record/<session>.json` and

      path = (
          explicit or runtime_dir.joinpath(*ACCOUNT_RECORD_DIRECTORY) / f"{session.isoformat()}.json"
      )
      if explicit is None and not path.is_file():
          return None
      if explicit is not None and not path.is_file():

              f"{session.isoformat()}, or omit the option"

          raise AccountRecordError(

def _refuse_a_shaped_history(root: Path, *, refusal: type[RuntimeError], consequence: str) -> None:
    """Refuse a repository whose history git reports reshaped -- shallow, or holding an
    `info/grafts` file -- naming the remedy and `consequence` (why the caller's question cannot
    be answered from the repository). Both entry points call this first: the guarantee is the
    function's, not its caller's order."""
    shallow = registry._git(root, "rev-parse", "--is-shallow-repository")
    if shallow.returncode != 0 or shallow.stdout.strip() != b"false":
        said = (shallow.stdout or shallow.stderr).decode(errors="replace").strip()
        raise refusal(
            f"{root} is a shallow repository, or git cannot say that it is not ({said!r}): its "
            f"history stops at the cut, so {consequence}; run `git fetch --unshallow` first"
        )
    grafts = registry._git(root, "rev-parse", "--git-path", "info/grafts")
    if grafts.returncode != 0:
        raise refusal(
            f"git cannot say where {root}'s grafts file would be "
            f"({grafts.stderr.decode(errors='replace').strip()}); {consequence}"
        )
    named = Path(grafts.stdout.decode().strip())
    grafts_file = named if named.is_absolute() else root / named
    if grafts_file.exists():
        raise refusal(
            f"{grafts_file} is a grafts file: it changes the parents git reports for every "
            f"reader (and the repository still says it is not shallow), so {consequence}; remove "
            "it, or convert it with `git replace --convert-graft-file`, whose replacement refs "
            "this module ignores"
        )
